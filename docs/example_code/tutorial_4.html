Tutorial 4 — NSHMP motions and FEM auto-profile project#

Motion selection (GMDB / NSHMP or user-defined spectrum), FEM project setup, optional automatic profile generation from a template project, compute, and result plots.

Requires: GMDB credentials for the motion-selection steps; NumPy and Matplotlib.

Download inputs (place under examples/Tutorial_4/Inputs/):

  • mean_profile_layers.csv

  • mean_profile_layers_column_map.csv

  • profile_generation_settings.csv

User-defined target spectrum (--target-source file; default path tests/resources/target_spectrum_example.txt):

  • target_spectrum_example.txt

For --randomize, provide an auto-profile template — see examples/Tutorial_4/template/README.md or tests/resources/Test_Auto_Profile.rsseismicfile.

 183def _motion_directory_registered(directories: list[str], output_directory: str) -> bool:
 184    target = os.path.normcase(os.path.abspath(output_directory))
 185    return any(os.path.normcase(os.path.abspath(path)) == target for path in directories)

 188def prepare_motion_output_directory(app: RSSeismicApplication, output_directory: str) -> str:
 189    output_directory = os.path.normpath(os.path.abspath(output_directory))
 190    os.makedirs(output_directory, exist_ok=True)
 191
 192    model = app.newProject()
 193    try:
 194        directories = model.Motions.listMotionDirectories()
 195        if _motion_directory_registered(directories, output_directory):
 196            print(f"  Motion folder already registered:\n    {output_directory}")
 197            return output_directory
 198
 199        print(f"  Registering motion folder:\n    {output_directory}")
 200        updated = model.Motions.addMotionDirectory(output_directory)
 201        if not _motion_directory_registered(updated, output_directory):
 202            raise RuntimeError(
 203                f"Could not register motion directory: {output_directory}\n"
 204                "Add it manually under RSSeismic Preferences → Motions."
 205            )
 206    finally:
 207        model.close(saveProject=False)
 208
 209    return output_directory

 282def resolve_target_spectra(
 283    library,
 284    target_source: str,
 285    target_file: str,
 286    *,
 287    nshmp: NshmpSiteParameters,
 288) -> TargetSpectra:
 289    source = target_source.strip().lower()
 290    if source == "file":
 291        path = os.path.normpath(target_file)
 292        if not os.path.isfile(path):
 293            raise FileNotFoundError(f"Target spectrum file not found: {path}")
 294        print(f"  Loading user-defined target from:\n    {path}")
 295        return library.getTargetSpectra(UserDefinedTargetSpectraModel(filePath=path))
 296
 297    if source == "nshmp":
 298        print("  Querying NSHMP Dynamic Hazard Curves...")
 299        print(f"  {_describe_nshmp_parameters(nshmp)}")
 300        return library.getTargetSpectra(_build_nshmp_target_spectra_model(nshmp))
 301
 302    raise ValueError(f"Unknown target source '{target_source}'. Use 'file' or 'nshmp'.")

 305def run_motion_selection_steps(
 306    app: RSSeismicApplication,
 307    *,
 308    motion_dir: str,
 309    database: GroundMotionDatabase,
 310    username: str,
 311    password: str,
 312    target_source: str,
 313    target_file: str,
 314    nshmp: NshmpSiteParameters,
 315    preview_max_results: int | None,
 316    overwrite: bool,
 317    plot_dir: str | None,
 318) -> TargetSpectra | None:
 319    if database == GroundMotionDatabase.CESMD:
 320        raise ValueError("Automatic selection supports GMDB and ESM only.")
 321    if not password.strip():
 322        raise ValueError("Database password is required.")
 323
 324    filters = _build_search_filters(MAGNITUDE_MIN, MAGNITUDE_MAX, RRUP_KM_MAX)
 325    selection = RecordSelectionParams(
 326        nGm=NUM_GROUND_MOTIONS,
 327        damping=SCALING_DAMPING,
 328        tolPercent=SCALING_TOLERANCE_PERCENT,
 329        isScaled=True,
 330    )
 331    amplitude = AmplitudeScalingParams(t1=AMPLITUDE_T1_S, t2=AMPLITUDE_T2_S)
 332    library = app.MotionLibrary
 333
 334    step(3, "Prepare motion output folder")
 335    motion_dir = prepare_motion_output_directory(app, motion_dir)
 336
 337    db_label = _database_label(database)
 338    step(4, f"Database: {db_label}")
 339    library.authenticateDatabase(database=database, username=username, password=password)
 340
 341    print("  Preview database search")
 342    preview = library.searchDatabaseRecords(
 343        database=database,
 344        filters=filters,
 345        maxResults=preview_max_results,
 346    )
 347    print(f"  Records returned: {len(preview)}")
 348    for record in preview[:5]:
 349        print(
 350            f"    {record.id}  M={record.magnitude}  Rrup={record.rrupKm:.1f} km  "
 351            f"sta={record.station}"
 352        )
 353    if len(preview) > 5:
 354        print(f"    ... and {len(preview) - 5} more")
 355
 356    step(
 357        5,
 358        "Target spectra (NSHMP Dynamic Hazard Curves)"
 359        if target_source.strip().lower() == "nshmp"
 360        else "Target spectra",
 361    )
 362    target = resolve_target_spectra(
 363        library, target_source, target_file, nshmp=nshmp,
 364    )
 365    if not target.points:
 366        raise RuntimeError("Target spectra returned no period–Sa points.")
 367    print(f"  {len(target.points)} period–Sa point(s).")
 368
 369    step(6, "Automatic record selection and save")
 370    print(
 371        "  Avoid opening the Motion tab before this step completes "
 372        "(or use --launch-rsseismic)."
 373    )
 374    try:
 375        result = library.matchToTargetSpectraAutomatic(
 376            database=database,
 377            filters=filters,
 378            targetSpectra=target,
 379            outputDirectory=motion_dir,
 380            selection=selection,
 381            amplitude=amplitude,
 382            overwrite=overwrite,
 383        )
 384    except ScriptingError as ex:
 385        if _is_motion_ui_collection_conflict(ex):
 386            print(_MOTION_UI_CONFLICT_HINT, file=sys.stderr)
 387        raise
 388
 389    print(f"  Saved {len(result.savedMotionNames)} motion component(s) under:\n    {motion_dir}")
 390    for name in result.savedMotionNames[:8]:
 391        print(f"    {name}")
 392    if len(result.savedMotionNames) > 8:
 393        print(f"    ... and {len(result.savedMotionNames) - 8} more")
 394
 395    if plot_dir is not None and result.savedMotionNames:
 396        step(7, "Plot target vs scaled geometric-mean spectra")
 397        period_arrays, sa_arrays = collect_geometric_mean_spectra(
 398            library, motion_dir, list(result.savedMotionNames),
 399        )
 400        if period_arrays:
 401            plot_path = os.path.join(os.path.abspath(plot_dir), MATCH_PLOT_FILENAME)
 402            plot_match_to_target_spectra(target, period_arrays, sa_arrays, plot_path)
 403            print(f"  Saved: {plot_path}")
 404        else:
 405            logger.warning("  No pair spectra available for match plot.")
 406
 407    return target

 425def _group_saved_motion_pairs(saved_motion_names: list[str]) -> list[tuple[str, str]]:
 426    buckets: dict[str, list[str]] = {}
 427    for motion_name in saved_motion_names:
 428        prefix = _shared_prefix_from_motion_name(motion_name)
 429        if prefix is None:
 430            continue
 431        buckets.setdefault(prefix, []).append(motion_name)
 432
 433    pairs: list[tuple[str, str]] = []
 434    for prefix, names in buckets.items():
 435        unique_names = sorted(set(names))
 436        if len(unique_names) == 2:
 437            pairs.append((unique_names[0], unique_names[1]))
 438    return pairs

 441def _pick_sa_column(spectral_df) -> str:
 442    for column in ("FrequencyDomain", "GeometricMeanSa", "Duhamel", "Newmark"):
 443        if column in spectral_df.columns:
 444            return column
 445    candidates = [col for col in spectral_df.columns if col != "Period"]
 446    if not candidates:
 447        raise ValueError(f"No spectral acceleration column in {list(spectral_df.columns)}")
 448    return candidates[0]

 451def _geometric_mean_spectrum_from_motion_names(
 452    library,
 453    h1_name: str,
 454    h2_name: str,
 455) -> tuple[np.ndarray, np.ndarray]:
 456    _, spec_h1 = library.getMotion(h1_name).getSpectralData()
 457    _, spec_h2 = library.getMotion(h2_name).getSpectralData()
 458    sa_column = _pick_sa_column(spec_h1)
 459
 460    period_h1 = spec_h1["Period"].values.astype(float)
 461    sa_h1 = spec_h1[sa_column].values.astype(float)
 462    period_h2 = spec_h2["Period"].values.astype(float)
 463    sa_h2 = spec_h2[sa_column].values.astype(float)
 464
 465    mask_h1 = np.isfinite(period_h1) & np.isfinite(sa_h1) & (period_h1 > 0) & (sa_h1 > 0)
 466    mask_h2 = np.isfinite(period_h2) & np.isfinite(sa_h2) & (period_h2 > 0) & (sa_h2 > 0)
 467    period_h1, sa_h1 = period_h1[mask_h1], sa_h1[mask_h1]
 468    period_h2, sa_h2 = period_h2[mask_h2], sa_h2[mask_h2]
 469
 470    order_h2 = np.argsort(period_h2)
 471    period_h2, sa_h2 = period_h2[order_h2], sa_h2[order_h2]
 472    sa_h2_on_h1 = np.interp(period_h1, period_h2, sa_h2, left=np.nan, right=np.nan)
 473    valid = np.isfinite(sa_h2_on_h1)
 474    period = period_h1[valid]
 475    geo_sa = np.sqrt(sa_h1[valid] * sa_h2_on_h1[valid])
 476    return period, geo_sa

 479def collect_geometric_mean_spectra(
 480    library,
 481    output_directory: str,
 482    saved_motion_names: list[str],
 483) -> tuple[list[np.ndarray], list[np.ndarray]]:
 484    periods_list: list[np.ndarray] = []
 485    sa_list: list[np.ndarray] = []
 486
 487    pairs = _group_saved_motion_pairs(saved_motion_names)
 488    if not pairs:
 489        discovered = library.discoverMotionPairsInFolder(output_directory)
 490        pairs = [(pair.motionH1Name, pair.motionH2Name) for pair in discovered.allPairs]
 491
 492    for h1_name, h2_name in pairs:
 493        try:
 494            period, geo_sa = _geometric_mean_spectrum_from_motion_names(
 495                library, h1_name, h2_name,
 496            )
 497        except Exception as ex:
 498            logger.warning("  Skipping pair for plot: %s", ex)
 499            continue
 500        if period.size:
 501            periods_list.append(period)
 502            sa_list.append(geo_sa)
 503
 504    return periods_list, sa_list

 507def _log_mean_spectrum(
 508    period_arrays: list[np.ndarray],
 509    sa_arrays: list[np.ndarray],
 510    common_periods: np.ndarray,
 511) -> np.ndarray:
 512    cx = np.log(common_periods)
 513    matrix = np.full((len(common_periods), len(period_arrays)), np.nan)
 514    for j, (periods, sa_values) in enumerate(zip(period_arrays, sa_arrays)):
 515        order = np.argsort(periods)
 516        xp = np.log(np.clip(periods[order], 1e-300, None))
 517        matrix[:, j] = np.interp(cx, xp, sa_values[order], left=np.nan, right=np.nan)
 518
 519    with np.errstate(divide="ignore", invalid="ignore"):
 520        ln_mean = np.nanmean(np.log(matrix), axis=1)
 521    return np.exp(ln_mean)

 524def plot_match_to_target_spectra(
 525    target: TargetSpectra,
 526    period_arrays: list[np.ndarray],
 527    sa_arrays: list[np.ndarray],
 528    plot_path: str,
 529) -> None:
 530    target_periods = np.array([point.period for point in target.points], dtype=float)
 531    target_sa = np.array([point.sa for point in target.points], dtype=float)
 532    target_mask = (
 533        np.isfinite(target_periods)
 534        & np.isfinite(target_sa)
 535        & (target_periods > 0)
 536        & (target_sa > 0)
 537    )
 538    target_periods = target_periods[target_mask]
 539    target_sa = target_sa[target_mask]
 540
 541    period_grid = np.sort(
 542        np.unique(np.concatenate([target_periods, period_arrays[0][period_arrays[0] > 0]]))
 543    )
 544    ensemble_sa = _log_mean_spectrum(period_arrays, sa_arrays, period_grid)
 545
 546    fig, ax = plt.subplots(figsize=(6, 4.5))

 693def log_mean_std(
 694    x_arrays: list[np.ndarray],
 695    y_arrays: list[np.ndarray],
 696    common_x: np.ndarray,
 697    *,
 698    log_x: bool = False,
 699    log_y: bool = True,
 700) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
 701    n_pts = len(common_x)
 702    n_samp = len(x_arrays)
 703    matrix = np.full((n_pts, n_samp), np.nan)
 704
 705    cx = np.log(common_x) if log_x else common_x
 706    for j, (x, y) in enumerate(zip(x_arrays, y_arrays)):
 707        order = np.argsort(x)
 708        xs, ys = x[order], y[order]
 709        xj = np.log(np.clip(xs, 1e-300, None)) if log_x else xs
 710        matrix[:, j] = np.interp(cx, xj, ys)
 711
 712    valid = np.isfinite(matrix) & (matrix > 0)
 713    if log_y:
 714        with np.errstate(divide="ignore", invalid="ignore"):
 715            ln_mat = np.where(valid, np.log(matrix), np.nan)
 716            mu = np.nanmean(ln_mat, axis=1)
 717            sigma = np.nanstd(ln_mat, axis=1, ddof=1)
 718        return np.exp(mu), np.exp(mu + sigma), np.exp(mu - sigma)
 719
 720    mu = np.nanmean(matrix, axis=1)
 721    sigma = np.nanstd(matrix, axis=1, ddof=1)
 722    return mu, mu + sigma, mu - sigma

 725def aggregate_by_layer_index(
 726    depth_arrays: list[np.ndarray],
 727    value_arrays: list[np.ndarray],
 728    *,
 729    log_y: bool = True,
 730) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
 731    max_layers = max(len(d) for d in depth_arrays)
 732    mean_depths: list[float] = []
 733    val_means: list[float] = []
 734    val_uppers: list[float] = []
 735    val_lowers: list[float] = []
 736
 737    for layer_idx in range(max_layers):
 738        depths: list[float] = []
 739        values: list[float] = []
 740        for depth_arr, value_arr in zip(depth_arrays, value_arrays):
 741            if layer_idx < len(depth_arr):
 742                depths.append(float(depth_arr[layer_idx]))
 743                values.append(float(value_arr[layer_idx]))
 744        if not values:
 745            continue
 746
 747        values_arr = np.array(values, dtype=float)
 748        if log_y:
 749            positive = values_arr[values_arr > 0]
 750            if positive.size == 0:
 751                continue
 752            ln_vals = np.log(positive)
 753            mu = float(np.mean(ln_vals))
 754            sigma = float(np.std(ln_vals, ddof=1)) if positive.size > 1 else 0.0
 755            val_means.append(math.exp(mu))
 756            val_uppers.append(math.exp(mu + sigma))
 757            val_lowers.append(math.exp(mu - sigma))
 758        else:
 759            mu = float(np.mean(values_arr))
 760            sigma = float(np.std(values_arr, ddof=1)) if values_arr.size > 1 else 0.0
 761            val_means.append(mu)
 762            val_uppers.append(mu + sigma)
 763            val_lowers.append(mu - sigma)
 764        mean_depths.append(float(np.mean(depths)))
 765
 766    return (
 767        np.array(mean_depths),
 768        np.array(val_means),
 769        np.array(val_uppers),
 770        np.array(val_lowers),
 771    )

1070def _set_mean_layer_shear_strength(layer, thickness_m: float, strength_kpa: float) -> None:
1071    """
1072    ``generateProfiles`` validates mean strata: shear strength must be > 0.
1073
1074    Uses ``MeanProfileSoilLayer`` (Constant_Value, else vary-with-depth table). Linear/Yee
1075    mean layers reject scalar GQ/H strength — vary-with-depth is attempted second.
1076    """
1077    if strength_kpa <= 0:
1078        raise ValueError(f"CSV shear_strength_kPa must be > 0 (got {strength_kpa}).")
1079
1080    thickness = max(thickness_m, 1e-6)
1081    last_error: Exception | None = None
1082
1083    for variation in ("Constant_Value", "Vary_With_Depth"):
1084        try:
1085            layer.setShearStrengthVariation(variation)
1086            if variation == "Constant_Value":
1087                layer.setShearStrength(strength_kpa)
1088            else:
1089                layer.setShearStrengthValues([(0.0, strength_kpa), (thickness, strength_kpa)])
1090            return
1091        except Exception as ex:
1092            last_error = ex
1093            continue
1094
1095    raise ScriptingError(
1096        "Could not set mean-profile shear strength for generateProfiles validation. "
1097        f"Last error: {last_error}"
